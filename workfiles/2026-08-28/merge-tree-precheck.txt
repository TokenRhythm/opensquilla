+"""sessions_fanout: synchronous fan-out of self-contained subtasks.
+
+Delegates 1-6 self-contained subtasks to subagents in waves (at most 3
+concurrent), waits for every child within the current turn via the task
+runtime, and returns one structured result set per tool call. The parent
+turn never ends while awaiting, so no cross-turn state or completion-push
+bookkeeping is required.
+
+Design: AGENTOS ``fanout鍘熻璇︾粏璁捐_v1.md`` (v1a).
+"""
+
+from __future__ import annotations
+
+import asyncio
+import json
+import time
+from pathlib import Path
+
+import structlog
+
+from opensquilla.session.models import AgentTaskStatus
+from opensquilla.tools.builtin.sessions import (
+    _get_session_manager,
+    _get_task_runtime,
+    _reject_guest_session_tool,
+    sessions_spawn,
+)
+from opensquilla.tools.registry import tool
+from opensquilla.tools.types import ToolError, current_tool_context
+
+_log = structlog.get_logger("opensquilla.tools.fanout")
+
+_MIN_TASKS = 1
+_MAX_TASKS = 6
+_DEFAULT_WAVE_SIZE = 3
+_MAX_WAVE_SIZE = 3
+_DEFAULT_TIMEOUT_SECONDS = 300
+_MAX_TIMEOUT_SECONDS = 900
+_TITLE_MAX_CHARS = 80
+_DESC_MAX_CHARS = 4000
+_OUTPUT_MAX_CHARS = 8000
+_TRANSCRIPT_READ_LIMIT = 50
+
+
+def _entry(
+    index: int,
+    title: str,
+    status: str,
+    *,
+    output: str | None = None,
+    error: str | None = None,
+    session_key: str | None = None,
+    output_truncated: bool = False,
+    output_file: str | None = None,
+) -> dict:
+    entry: dict = {
+        "index": index,
+        "title": title,
+        "status": status,
+        "output": output,
+        "output_truncated": output_truncated,
+        "output_file": output_file,
+        "session_key": session_key,
+    }
+    if error is not None:
+        entry["error"] = error
+    return entry
+
+
+def _row_value(row: object, key: str) -> object:
+    if isinstance(row, dict):
+        return row.get(key)
+    return getattr(row, key, None)
+
+
+def _content_to_text(content: object) -> str:
+    if isinstance(content, str):
+        return content
+    if isinstance(content, list):
+        parts: list[str] = []
+        for part in content:
+            if isinstance(part, str):
+                parts.append(part)
+                continue
+            text = part.get("text") if isinstance(part, dict) else getattr(part, "text", None)
+            if isinstance(text, str):
+                parts.append(text)
+        return "".join(parts)
+    if content is None:
+        return ""
+    return str(content)
+
+
+async def _read_final_assistant_text(session_key: str) -> str:
+    """Return the last non-empty assistant message of a child transcript."""
+
+    mgr = _get_session_manager()
+    read_transcript = getattr(mgr, "read_transcript", None)
+    if not callable(read_transcript):
+        return ""
+    try:
+        rows = await read_transcript(session_key, limit=_TRANSCRIPT_READ_LIMIT)
+    except Exception:
+        return ""
+    for row in reversed(list(rows or [])):
+        if _row_value(row, "role") != "assistant":
+            continue
+        text = _content_to_text(_row_value(row, "content")).strip()
+        if text:
+            return text
+    return ""
+
+
+def _terminal_error_text(record: object) -> str:
+    parts: list[str] = []
+    for key in ("error_class", "error_message", "terminal_reason"):
+        value = getattr(record, key, None)
+        if isinstance(value, str) and value.strip():
+            parts.append(value.strip())
+    return ": ".join(parts)
+
+
+def _spill_output(full_text: str, out_dir: Path, index: int) -> str | None:
+    try:
+        out_dir.mkdir(parents=True, exist_ok=True)
+        path = out_dir / f"{index:02d}.md"
+        path.write_text(full_text, encoding="utf-8")
+        return str(path)
+    except OSError:
+        return None
+

+@tool(
+    name="sessions_fanout",
+    description=(
+        "Delegate 1-6 self-contained subtasks to subagents in parallel waves and "
+        "wait synchronously for all of them, returning structured results in a "
+        "single call. Use for work that decomposes cleanly (multi-source "
+        "empirical checks, multi-file processing, multi-entity comparison). Each "
+        "task must be self-contained (desc+expect): subagents cannot see the "
+        "parent conversation, sibling tasks, or acceptance criteria. Do not call "
+        "this tool for a single lightweight task; executing it directly is "
+        "faster."
+    ),
+    params={
+        "tasks": {
+            "type": "array",
+            "description": (
+                "1-6 self-contained subtasks. Subagents cannot see the parent "
+                "conversation, sibling tasks, or each other."
+            ),
+            "items": {
+                "type": "object",
+                "properties": {
+                    "title": {
+                        "type": "string",
+                        "description": "Short subtask title (max 80 chars).",
+                    },
+                    "desc": {
+                        "type": "string",
+                        "description": (
+                            "Complete self-contained instruction for this subtask "
+                            "(max 4000 chars). It has no other context."
+                        ),
+                    },
+                    "expect": {
+                        "type": "string",
+                        "description": (
+                            "Verifiable description of the expected output, used "
+                            "by the parent to accept or re-run. Not shown to the "
+                            "subagent."
+                        ),
+                    },
+                },
+                "required": ["title", "desc", "expect"],
+            },
+        },
+        "model": {
+            "type": "string",
+            "description": (
+                "Optional worker model override (must resolve under the parent "
+                "provider). Omit to inherit the parent model."
+            ),
+        },
+        "wave_size": {
+            "type": "integer",
+            "description": "Concurrent subagents per wave (default 3, hard cap 3).",
+            "default": _DEFAULT_WAVE_SIZE,
+        },
+        "timeout_seconds": {
+            "type": "integer",
+            "description": (
+                "Total deadline for all waves in seconds (default 300, hard cap 900)."
+            ),
+            "default": _DEFAULT_TIMEOUT_SECONDS,
+        },
+    },
+    required=["tasks"],
+    execution_timeout_argument="timeout_seconds",
+    execution_timeout_padding=15.0,
+)
+async def sessions_fanout(
+    tasks: list[dict] | None = None,
+    model: str | None = None,
+    wave_size: int = _DEFAULT_WAVE_SIZE,
+    timeout_seconds: int = _DEFAULT_TIMEOUT_SECONDS,
+) -> str:
+    _reject_guest_session_tool("sessions_fanout")
+
+    # 鈹€鈹€ Whole-call validation: fail fast before spawning anything 鈹€鈹€
+    if not isinstance(tasks, list):
+        raise ToolError("tasks must be an array of subtask objects")
+    if not (_MIN_TASKS <= len(tasks) <= _MAX_TASKS):
+        raise ToolError(
+            f"tasks must contain {_MIN_TASKS}-{_MAX_TASKS} items (got {len(tasks)})"
+        )
+    items: list[tuple[str, str, str]] = []
+    for i, item in enumerate(tasks):
+        if not isinstance(item, dict):
+            raise ToolError(f"tasks[{i}] must be an object")
+        title = str(item.get("title") or "").strip()
+        desc = str(item.get("desc") or "").strip()
+        expect = str(item.get("expect") or "").strip()
+        if not title:
+            raise ToolError(f"tasks[{i}].title is required")
+        if len(title) > _TITLE_MAX_CHARS:
+            raise ToolError(f"tasks[{i}].title exceeds {_TITLE_MAX_CHARS} characters")
+        if not desc:
+            raise ToolError(f"tasks[{i}].desc is required")
+        if len(desc) > _DESC_MAX_CHARS:
+            raise ToolError(f"tasks[{i}].desc exceeds {_DESC_MAX_CHARS} characters")
+        if not expect:
+            raise ToolError(f"tasks[{i}].expect is required")
+        items.append((title, desc, expect))
+
+    try:
+        wave = max(1, min(int(wave_size), _MAX_WAVE_SIZE))
+        budget = max(1, min(int(timeout_seconds), _MAX_TIMEOUT_SECONDS))
+    except (TypeError, ValueError):
+        raise ToolError("wave_size and timeout_seconds must be integers")
+
+    ctx = current_tool_context.get()
+    parent_session_key = getattr(ctx, "session_key", None) if ctx is not None else None
+    parent_task_id = getattr(ctx, "task_id", None) if ctx is not None else None
+    if not parent_session_key or not parent_task_id:
+        raise ToolError("sessions_fanout requires an active parent turn context")
+    parent_session_key = str(parent_session_key)
+    parent_task_id = str(parent_task_id)
+
+    try:
+        runtime = _get_task_runtime()
+    except ToolError:
+        raise ToolError("Task runtime not available; sessions_fanout cannot wait")
+
+    # Suppress the per-child background parent-wake for this task group:
+    # results are delivered synchronously through this tool return, so a
+    # later synthesis wake turn would be a phantom turn. Children still run
+    # and their completions are still recorded to the parent transcript.
+    try:
+        from opensquilla.gateway.subagent_announce import (
+            cancel_background_completion_for_task,
+        )
+
+        await cancel_background_completion_for_task(parent_session_key, parent_task_id)
+    except Exception:
+        pass  # best-effort; an unclosed spawn group also never wakes the parent
+
+    started = time.monotonic()
+    deadline = started + budget
+    n = len(items)
+    results: list[dict] = [None] * n
+    timed_out = False
+    waves_run = 0
+
+    async def _run_one_impl(index: int) -> None:
+        nonlocal timed_out
+        title, desc, _expect = items[index]
+        try:
+            spawned = json.loads(await sessions_spawn(task=desc, model=model, title=title))
+        except Exception as exc:
+            results[index] = _entry(index, title, "failed", error=f"spawn failed: {exc}")
+            return
+        child_session_key = str(spawned.get("session_key") or "") or None
+        child_task_id = str(spawned.get("task_id") or "")
+        remaining = deadline - time.monotonic()
+        if remaining <= 0:
+            timed_out = True
+            results[index] = _entry(
+                index,
+                title,
+                "timeout",
+                session_key=child_session_key,
+                error="deadline reached before waiting began; child left running",
+            )
+            return
+        try:
+            record = await runtime.wait(child_task_id, timeout=remaining)
+        except TimeoutError:
+            timed_out = True
+            results[index] = _entry(
+                index,
+                title,
+                "timeout",
+                session_key=child_session_key,
+                error="deadline reached; child task left running in the background",
+            )
+            return
+        except Exception as exc:
+            results[index] = _entry(
+                index,
+                title,
+                "failed",
+                session_key=child_session_key,
+                error=f"wait failed: {exc}",
+            )
+            return
+        status = getattr(record, "status", None)
+        if status == AgentTaskStatus.SUCCEEDED:
+            text = await _read_final_assistant_text(child_session_key)
+            if not text.strip():
+                results[index] = _entry(
+                    index,
+                    title,
+                    "failed",
+                    session_key=child_session_key,
+                    error="(empty) subagent produced no assistant text",
+                )
+                return
+            truncated = len(text) > _OUTPUT_MAX_CHARS
+            output_file: str | None = None
+            if truncated:
+                output = text[:_OUTPUT_MAX_CHARS]
+                workspace_dir = getattr(ctx, "workspace_dir", None)
+                if isinstance(workspace_dir, str) and workspace_dir.strip():
+                    output_file = _spill_output(
+                        text,
+                        Path(workspace_dir) / "workfiles" / "fanout" / parent_task_id,
+                        index,
+                    )
+            else:
+                output = text
+            results[index] = _entry(
+                index,
+                title,
+                "ok",
+                session_key=child_session_key,
+                output=output,
+                output_truncated=truncated,
+                output_file=output_file,
+            )
+            return
+        if status == AgentTaskStatus.TIMEOUT:
+            timed_out = True
+            results[index] = _entry(
+                index,
+                title,
+                "timeout",
+                session_key=child_session_key,
+                error=_terminal_error_text(record) or "child task timed out",
+            )
+            return
+        results[index] = _entry(
+            index,
+            title,
+            "failed",
+            session_key=child_session_key,
+            error=_terminal_error_text(record)
+            or f"child task ended with status={getattr(status, 'value', status)}",
+        )
+
+    async def _run_one(index: int) -> None:
+        try:
+            await _run_one_impl(index)
+        except Exception as exc:  # defensive: one child must never kill the round
+            if results[index] is None:
+                results[index] = _entry(
+                    index, items[index][0], "failed", error=f"internal error: {exc}"
+                )
+
+    for wave_start in range(0, n, wave):
+        if time.monotonic() >= deadline:
+            timed_out = True
+            for i in range(wave_start, n):
+                results[i] = _entry(
+                    i,
+                    items[i][0],
+                    "timeout",
+                    error="deadline reached before this wave started",
+                )
+            break
+        waves_run += 1
+        wave_indices = list(range(wave_start, min(wave_start + wave, n)))
+        await asyncio.gather(*(_run_one(i) for i in wave_indices))
+
+    for i, entry in enumerate(results):
+        if entry is None:
+            results[i] = _entry(
+                i, items[i][0], "failed", error="internal error: no result"
+            )
+
+    n_ok = sum(1 for r in results if r["status"] == "ok")
+    n_failed = sum(1 for r in results if r["status"] == "failed")
+    n_timeout = sum(1 for r in results if r["status"] == "timeout")
+    duration = round(time.monotonic() - started, 1)
+    _log.info(
+        "sessions_fanout.completed",
+        n_total=n,
+        n_ok=n_ok,
+        n_failed=n_failed,
+        n_timeout=n_timeout,
+        waves=waves_run,
+        duration_seconds=duration,
+        parent_session_key=parent_session_key,
+    )
+    return json.dumps(
+        {
+            "summary": {
+                "n_total": n,
+                "n_ok": n_ok,
+                "n_failed": n_failed,
+                "n_timeout": n_timeout,
+                "waves": waves_run,
+                "duration_seconds": duration,
+                "timed_out": timed_out,
+            },
+            "results": results,
+        },
+        ensure_ascii=False,
+    )
